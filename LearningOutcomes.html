Database Creation

Connecting to a MySQL database and creating an Employee table using Node.js.

create-table.js
var mysql = require('mysql');
var con = mysql
  .createConnection({
    host: "127.0.0.1",
    user: "root",
    password: "",
    database: "Texury"
  });

con.connect(function(err){
  if(err) throw err;
  console.log("connected!");
  var sql = "CREATE TABLE
    Employee (eid int(5),
    age int(5),
    name varchar(30))"
  con.query(sql,
    function(err, result){
      if(err) throw err;
      console.log(
        "Table created");
  });
});

03 ·

Button Hover Function

A hover function where the button changes colour dynamically using JavaScript event listeners.

hover-button.js
const btn = document
  .querySelector('button');

btn.addEventListener(
  'mouseover',
  function() {
    btn.style
      .backgroundColor
      = '#ff9900';
    btn.style.color
      = '#000000';
});

btn.addEventListener(
  'mouseout',
  function() {
    btn.style
      .backgroundColor
      = '#000000';
    btn.style.color
      = '#ffffff';
});
